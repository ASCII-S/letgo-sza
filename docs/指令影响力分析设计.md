# 基于数据流和控制流的指令影响力分析

本文档设计一套在故障注入前分析指令影响能力的方案，通过数据流图(DFG)和控制流图(CFG)评估指令的影响范围。

---

## 1. 核心概念

### 1.1 影响力定义

```
指令影响力 = f(数据流传播范围, 控制流影响范围, 传播深度, 关键路径权重)
```

**影响力类型**：

| 类型 | 描述 | 示例 |
|------|------|------|
| **数据影响** | 值传播到多少后续指令 | `rax` 被 10 条后续指令使用 |
| **控制影响** | 影响多少分支决策 | 条件跳转决定循环次数 |
| **内存影响** | 写入的内存被多少指令读取 | 数组元素被后续循环访问 |
| **调用影响** | 影响函数调用的参数/返回值 | 参数寄存器传播到被调用函数 |

### 1.2 分析方法对比

| 方法 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| **静态分析** | 完整覆盖、无运行开销 | 无法处理间接跳转/动态内存 | CFG 构建、基本块分析 |
| **动态分析** | 精确、处理动态行为 | 路径覆盖有限、运行开销 | 数据流追踪、污点分析 |
| **混合分析** | 结合两者优点 | 实现复杂 | 完整影响力评估 |

## 2. 静态分析：控制流图 (CFG)

### 2.1 CFG 构建工具

使用 `objdump` + Python 解析，或 Pin 工具提取：

```python
# cfg_analyzer.py

import subprocess
import re
from collections import defaultdict

class CFGAnalyzer:
    """控制流图分析器"""

    def __init__(self, binary_path):
        self.binary = binary_path
        self.cfg = defaultdict(list)  # 邻接表表示
        self.basic_blocks = {}        # 基本块信息
        self.dominators = {}          # 支配关系

    def build_cfg_from_objdump(self):
        """从 objdump 输出构建 CFG"""
        # 获取反汇编
        result = subprocess.run(
            ['objdump', '-d', self.binary],
            capture_output=True, text=True
        )
        asm_lines = result.stdout.split('\n')

        current_func = None
        current_block = None
        prev_addr = None

        for line in asm_lines:
            # 匹配函数头
            func_match = re.match(r'^([0-9a-f]+)\s+<(\w+)>:', line)
            if func_match:
                current_func = func_match.group(2)
                current_block = int(func_match.group(1), 16)
                continue

            # 匹配指令
            inst_match = re.match(r'^\s+([0-9a-f]+):\s+.*?\s+(\w+)\s*(.*)', line)
            if inst_match:
                addr = int(inst_match.group(1), 16)
                mnemonic = inst_match.group(2)
                operands = inst_match.group(3)

                # 检测基本块边界
                if self._is_control_flow(mnemonic):
                    # 添加 CFG 边
                    target = self._extract_target(mnemonic, operands)
                    if target:
                        self.cfg[current_block].append(target)

                    # 顺序执行边（非无条件跳转）
                    if mnemonic not in ['jmp', 'ret']:
                        next_addr = addr + self._inst_size(line)
                        self.cfg[current_block].append(next_addr)

                    # 新建基本块
                    current_block = addr + self._inst_size(line)

                prev_addr = addr

    def _is_control_flow(self, mnemonic):
        """判断是否为控制流指令"""
        cf_instructions = [
            'jmp', 'je', 'jne', 'jz', 'jnz', 'jl', 'jle', 'jg', 'jge',
            'ja', 'jae', 'jb', 'jbe', 'call', 'ret', 'loop', 'loope', 'loopne'
        ]
        return mnemonic.lower() in cf_instructions

    def _extract_target(self, mnemonic, operands):
        """提取跳转目标地址"""
        # 直接地址: jmp 0x401234
        direct = re.search(r'([0-9a-f]+)', operands)
        if direct:

    def find_control_dependence(self, inst_addr):
        """
        找出依赖于该指令的所有基本块

        如果 inst_addr 是条件跳转，返回所有受其控制的基本块
        """
        dependent_blocks = set()

        # 找到包含该指令的基本块
        block = self._find_block(inst_addr)

        # BFS 遍历所有可达的后继块
        visited = set()
        queue = [block]

        while queue:
            current = queue.pop(0)
            if current in visited:
                continue
            visited.add(current)
            dependent_blocks.add(current)

            for successor in self.cfg[current]:
                queue.append(successor)

        return dependent_blocks

    def calculate_control_influence(self, inst_addr):
        """
        计算控制流影响力分数

        分数 = 受控制的基本块数量 × 块平均大小 × 路径权重
        """
        dependent = self.find_control_dependence(inst_addr)
        return len(dependent)
```

## 3. 动态分析：数据流追踪

### 3.1 基于 Pin 的污点分析工具

```cpp
// dataflow_tracker.cpp - Pin 工具实现动态数据流追踪

#include "pin.H"
#include <map>
#include <set>
#include <vector>
#include <fstream>

// 污点标签
typedef UINT32 TaintTag;

// 寄存器污点状态
std::map<REG, TaintTag> reg_taint;

// 内存污点状态
std::map<ADDRINT, TaintTag> mem_taint;

// 数据流边记录: source_tag -> {dest_instructions}
std::map<TaintTag, std::set<ADDRINT>> dataflow_edges;

// 当前污点 ID
TaintTag current_tag = 0;

// 目标注入点
ADDRINT target_inject_pc = 0;
REG target_inject_reg;
bool tracking_enabled = false;

// ============ 污点传播规则 ============

VOID PropagateRegToReg(ADDRINT pc, REG src, REG dst) {
    if (!tracking_enabled) return;

    if (reg_taint.count(src)) {
        TaintTag tag = reg_taint[src];
        reg_taint[dst] = tag;
        dataflow_edges[tag].insert(pc);
    }
}

VOID PropagateMemToReg(ADDRINT pc, ADDRINT mem_addr, REG dst) {
    if (!tracking_enabled) return;

    if (mem_taint.count(mem_addr)) {
        TaintTag tag = mem_taint[mem_addr];
        reg_taint[dst] = tag;
        dataflow_edges[tag].insert(pc);
    }
}

VOID PropagateRegToMem(ADDRINT pc, REG src, ADDRINT mem_addr) {
    if (!tracking_enabled) return;

    if (reg_taint.count(src)) {
        TaintTag tag = reg_taint[src];
        mem_taint[mem_addr] = tag;
        dataflow_edges[tag].insert(pc);
    }
}

// ============ 指令分析插桩 ============

VOID InstrumentInstruction(INS ins, VOID* v) {
    ADDRINT pc = INS_Address(ins);

    // 检查是否到达目标注入点
    if (pc == target_inject_pc) {
        INS_InsertCall(ins, IPOINT_AFTER, (AFUNPTR)StartTracking,
                       IARG_UINT32, target_inject_reg,
                       IARG_END);
    }

    // 寄存器到寄存器
    if (INS_RegRContain(ins, target_inject_reg) && INS_MaxNumWRegs(ins) > 0) {
        REG dst = INS_RegW(ins, 0);
        INS_InsertCall(ins, IPOINT_BEFORE, (AFUNPTR)PropagateRegToReg,
                       IARG_INST_PTR,
                       IARG_UINT32, target_inject_reg,
                       IARG_UINT32, dst,
                       IARG_END);
    }

    // 内存读取
    if (INS_IsMemoryRead(ins)) {
        REG dst = INS_RegW(ins, 0);
        INS_InsertCall(ins, IPOINT_BEFORE, (AFUNPTR)PropagateMemToReg,
                       IARG_INST_PTR,
                       IARG_MEMORYREAD_EA,
                       IARG_UINT32, dst,
                       IARG_END);
    }

    // 内存写入
    if (INS_IsMemoryWrite(ins)) {
        for (UINT32 i = 0; i < INS_MaxNumRRegs(ins); i++) {
            REG src = INS_RegR(ins, i);
            INS_InsertCall(ins, IPOINT_BEFORE, (AFUNPTR)PropagateRegToMem,
                           IARG_INST_PTR,
                           IARG_UINT32, src,
                           IARG_MEMORYWRITE_EA,
                           IARG_END);
        }
    }
}

VOID StartTracking(REG reg) {
    tracking_enabled = true;
    current_tag++;
    reg_taint[reg] = current_tag;
    std::cout << "Started tracking from reg " << REG_StringShort(reg)
              << " with tag " << current_tag << std::endl;
}

VOID Fini(INT32 code, VOID* v) {
    // 输出数据流图
    std::ofstream out("dataflow_graph.csv");
    out << "source_tag,influenced_pc,influenced_count" << std::endl;

    for (auto& kv : dataflow_edges) {
        TaintTag tag = kv.first;
        std::set<ADDRINT>& influenced = kv.second;

        out << tag << ",";
        for (ADDRINT pc : influenced) {
            out << std::hex << pc << ";";
        }
        out << "," << influenced.size() << std::endl;
    }
    out.close();

    std::cout << "Data flow analysis complete. "
              << "Total influenced instructions: "
              << dataflow_edges[current_tag].size() << std::endl;
}

int main(int argc, char* argv[]) {
    PIN_Init(argc, argv);

    // 解析参数
    target_inject_pc = KnobTargetPC.Value();
    target_inject_reg = REG_StringToReg(KnobTargetReg.Value());

    INS_AddInstrumentFunction(InstrumentInstruction, 0);
    PIN_AddFiniFunction(Fini, 0);

    PIN_StartProgram();
    return 0;
}
```

```python
import subprocess
import csv
import os
import configure

class DataFlowAnalyzer:
    """数据流影响力分析器"""

    def __init__(self):
        self.pin_tool = os.path.join(
            configure.toolbase, "obj-intel64/dataflow_tracker.so"
        )
        self.results = {}

    def analyze_injection_point(self, pc, reg, max_instructions=10000):
        """
        分析指定注入点的数据流影响

        Args:
            pc: 注入点 PC 地址
            reg: 注入的目标寄存器
            max_instructions: 最大追踪指令数

        Returns:
            DataFlowResult 对象
        """
        # 运行 Pin 工具
        execlist = [
            configure.pin_home, '-t', self.pin_tool,
            '-target_pc', str(pc),
            '-target_reg', reg,
            '-max_inst', str(max_instructions),
            '--', configure.benchmark
        ]
        execlist.extend(configure.args)

        subprocess.run(execlist, capture_output=True)

        # 解析结果
        return self._parse_dataflow_result('dataflow_graph.csv')

    def _parse_dataflow_result(self, filepath):
        """解析数据流分析结果"""
        result = DataFlowResult()

        with open(filepath, 'r') as f:
            reader = csv.DictReader(f)
            for row in reader:
                influenced_pcs = row['influenced_pc'].split(';')
                influenced_pcs = [int(pc, 16) for pc in influenced_pcs if pc]
                result.influenced_instructions = influenced_pcs
                result.influence_count = int(row['influenced_count'])

        return result


class DataFlowResult:
    """数据流分析结果"""

    def __init__(self):
        self.influenced_instructions = []  # 受影响的指令地址列表
        self.influence_count = 0           # 受影响指令数量
        self.memory_writes = []            # 受影响的内存写入
        self.control_flow_deps = []        # 影响的控制流指令

    def get_influence_score(self):
        """计算数据流影响力分数"""
        base_score = min(self.influence_count / 100, 10)

        # 如果影响了控制流指令，加权
        cf_bonus = len(self.control_flow_deps) * 0.5

        # 如果影响了内存写入，加权
        mem_bonus = len(self.memory_writes) * 0.3

        return min(base_score + cf_bonus + mem_bonus, 10)
```

## 4. 综合影响力评分模型

### 4.1 影响力评分公式

```python
# influence_scorer.py

class InstructionInfluenceScorer:
    """指令影响力综合评分器"""

    def __init__(self, cfg_analyzer, dataflow_analyzer):
        self.cfg = cfg_analyzer
        self.dfa = dataflow_analyzer

        # 权重配置
        self.weights = {
            'dataflow': 0.35,      # 数据流影响权重
            'control_flow': 0.30,  # 控制流影响权重
            'memory': 0.20,        # 内存影响权重
            'criticality': 0.15   # 路径关键性权重
        }

    def calculate_influence_score(self, inst_info):
        """
        计算指令的综合影响力分数

        Args:
            inst_info: dict with keys: pc, reg, mnemonic

        Returns:
            float: 0-10 的影响力分数
        """
        pc = inst_info['pc']
        reg = inst_info.get('reg') or inst_info.get('regmm')

        scores = {}

        # 1. 数据流影响分数
        if reg:
            df_result = self.dfa.analyze_injection_point(pc, reg)
            scores['dataflow'] = df_result.get_influence_score()
        else:
            scores['dataflow'] = 0

        # 2. 控制流影响分数
        scores['control_flow'] = self._calc_control_flow_score(pc, inst_info)

        # 3. 内存影响分数
        scores['memory'] = self._calc_memory_score(inst_info)

        # 4. 路径关键性分数
        scores['criticality'] = self._calc_criticality_score(pc)

        # 加权求和
        total_score = sum(
            scores[k] * self.weights[k] for k in self.weights
        )

        return {
            'total_score': round(total_score, 2),
            'breakdown': scores,
            'influenced_count': df_result.influence_count if reg else 0
        }

    def _calc_control_flow_score(self, pc, inst_info):
        """计算控制流影响分数"""
        mnemonic = inst_info.get('mnemonic', '').lower()

        # 控制流指令本身得分高
        if mnemonic in ['call', 'ret', 'jmp']:
            base = 8
        elif mnemonic.startswith('j'):  # 条件跳转
            base = 7
        else:
            base = 2

        # 加上可达块数量的加成
        reachable = self.cfg.calculate_control_influence(pc)
        reach_bonus = min(reachable / 10, 2)

        return min(base + reach_bonus, 10)

    def _calc_memory_score(self, inst_info):
        """计算内存影响分数"""
        is_mem_write = inst_info.get('is_memory_write', False)
        is_stack = inst_info.get('is_stack_op', False)

        if is_mem_write and is_stack:
            return 9  # 栈写入，高风险
        elif is_mem_write:
            return 7  # 普通内存写入
        elif is_stack:
            return 4  # 栈读取
        else:
            return 1  # 非内存操作

    def _calc_criticality_score(self, pc):
        """计算路径关键性分数"""
        # 检查是否在循环内
        loop_depth = self.cfg.loop_depth(pc) if hasattr(self.cfg, 'loop_depth') else 0

        # 检查是否为循环条件
        is_loop_cond = self.cfg.is_loop_condition(pc) if hasattr(self.cfg, 'is_loop_condition') else False

        score = 3  # 基础分
        score += min(loop_depth * 2, 4)  # 循环深度加成
        if is_loop_cond:
            score += 3  # 循环条件加成

        return min(score, 10)
```

### 4.2 批量分析与排序

```python
def analyze_and_rank_instructions(catalog_file, output_file, top_n=100):
    """
    分析 catalog 中所有指令的影响力并排序

    Args:
        catalog_file: 输入的指令 catalog
        output_file: 输出的排序结果
        top_n: 返回前 N 条高影响力指令
    """
    import pandas as pd

    # 初始化分析器
    cfg = CFGAnalyzer(configure.benchmark)
    cfg.build_cfg_from_objdump()

    dfa = DataFlowAnalyzer()
    scorer = InstructionInfluenceScorer(cfg, dfa)

    # 读取 catalog
    df = pd.read_csv(catalog_file)

    # 为每条指令计算影响力分数
    results = []
    for idx, row in df.iterrows():
        inst_info = {
            'pc': row['pc'],
            'reg': row.get('reg', ''),
            'regmm': row.get('regmm', ''),
            'mnemonic': row.get('mnemonic', ''),
            'is_memory_write': row.get('is_memory_write', False),
            'is_stack_op': row.get('is_stack_op', False),
        }

        score_result = scorer.calculate_influence_score(inst_info)

        results.append({
            'pc': row['pc'],
            'reg': row.get('reg', ''),
            'regmm': row.get('regmm', ''),
            'mnemonic': row.get('mnemonic', ''),
            'influence_score': score_result['total_score'],
            'dataflow_score': score_result['breakdown']['dataflow'],
            'control_score': score_result['breakdown']['control_flow'],
            'memory_score': score_result['breakdown']['memory'],
            'criticality_score': score_result['breakdown']['criticality'],
            'influenced_count': score_result['influenced_count'],
        })

        if idx % 100 == 0:
            print(f"Analyzed {idx}/{len(df)} instructions...")

    # 按影响力分数排序
    result_df = pd.DataFrame(results)
    result_df = result_df.sort_values('influence_score', ascending=False)

    # 保存结果
    result_df.to_csv(output_file, index=False)
    print(f"Saved {len(result_df)} instructions to {output_file}")

    # 返回 top N
    return result_df.head(top_n)
```

```python
## 影响力分布统计

### 按指令类型
"""

    # 按指令类型统计
    type_stats = analysis_results.groupby('mnemonic')['influence_score'].agg(['mean', 'max', 'count'])
    type_stats = type_stats.sort_values('mean', ascending=False)

    report += "| 指令类型 | 平均影响力 | 最大影响力 | 数量 |\n"
    report += "|----------|-----------|-----------|------|\n"
    for mnemonic, row in type_stats.head(15).iterrows():
        report += f"| {mnemonic} | {row['mean']:.2f} | {row['max']:.2f} | {int(row['count'])} |\n"

    with open(output_path, 'w') as f:
        f.write(report)

    print(f"Report saved to {output_path}")
```

```python
def generate_influence_weighted_pool(
    catalog_file,
    pool_file,
    num_samples=2000,
    influence_threshold=5.0
):
    """
    生成基于影响力加权的指令池

    优先选择高影响力指令进行故障注入
    """
    # 1. 运行影响力分析
    influence_results = analyze_and_rank_instructions(
        catalog_file,
        'influence_analysis.csv'
    )

    # 2. 过滤低影响力指令
    high_influence = influence_results[
        influence_results['influence_score'] >= influence_threshold
    ]

    print(f"Found {len(high_influence)} high-influence instructions")

    # 3. 按影响力分数作为权重采样
    weights = high_influence['influence_score'].values
    weights = weights / weights.sum()

    selected = high_influence.sample(
        n=min(num_samples, len(high_influence)),
        weights=weights,
        replace=True
    )

    # 4. 生成 pool 文件
    pool_data = []
    for _, row in selected.iterrows():
        pool_data.append([
            row['regmm'] if row['regmm'] else '',
            row['reg'] if row['reg'] else '',
            str(row['pc']),
            str(random.randint(0, 1024))
        ])

    with open(pool_file, 'w', newline='') as f:
        writer = csv.writer(f)
        writer.writerows(pool_data)

    print(f"Generated {len(pool_data)} injection points based on influence analysis")
    return pool_file
```

```python
# 影响力-结果相关性分析
def evaluate_influence_correlation(results_df):
    """评估影响力分数与实验结果的相关性"""
    from scipy.stats import pearsonr, spearmanr

    # 将结果转为数值
    result_map = {
        'Masked': 0,
        'SDC': 1,
        'C-Masked': 2,
        'C-SDC': 2,
        'Recrash': 3
    }
    results_df['result_score'] = results_df['result'].map(result_map)

    # 计算相关系数
    pearson_corr, _ = pearsonr(
        results_df['influence_score'],
        results_df['result_score']
    )
    spearman_corr, _ = spearmanr(
        results_df['influence_score'],
        results_df['result_score']
    )

    print(f"Pearson correlation: {pearson_corr:.3f}")
    print(f"Spearman correlation: {spearman_corr:.3f}")

    return pearson_corr, spearman_corr
```
