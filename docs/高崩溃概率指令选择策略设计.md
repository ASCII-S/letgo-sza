# 高崩溃概率指令选择策略设计

本文档设计一套基于崩溃概率的指令选择策略，用于优化故障注入实验的效率。

---

## 1. 问题分析

### 1.1 当前框架的指令选择方式

| 模式 | 选择方式 | 代码位置 |
|------|----------|----------|
| `random` | 完全随机选择动态指令 | `faultinject.py:getBreakpoint` |
| `targeted` | 按指令类型选择（如 `div`, `mov`, `stack`） | `InstPoolMaker.py:generate_catalog` |

### 1.2 崩溃概率的影响因素

```
崩溃概率 = f(指令类型, 操作数类型, 程序位置, 寄存器敏感性, 内存访问模式)
```

**高崩溃概率的指令特征**：
1. **内存写操作**：错误地址导致 SIGSEGV
2. **控制流指令**：跳转到非法地址
3. **栈操作指令**：破坏栈帧结构
4. **指针运算指令**：产生非法指针
5. **关键寄存器操作**：rsp, rbp, rip 相关

---

## 2. 设计方案

### 2.1 多维度崩溃概率评分模型

```
CrashScore = w1 * TypeScore + w2 * RegScore + w3 * MemScore + w4 * HistoryScore
```

| 维度 | 权重 | 说明 |
|------|------|------|
| TypeScore | w1=0.3 | 指令类型敏感性评分 |
| RegScore | w2=0.25 | 寄存器敏感性评分 |
| MemScore | w3=0.25 | 内存操作模式评分 |
| HistoryScore | w4=0.2 | 历史崩溃率统计 |

### 2.2 各维度评分规则

#### 2.2.1 指令类型敏感性 (TypeScore)

```python
# 基于指令类型的崩溃敏感性评分 (0-10分)
INST_TYPE_CRASH_SCORE = {
    # 高敏感 (8-10分)
    'call': 10,      # 调用指令，修改返回地址极易崩溃
    'ret': 10,       # 返回指令，栈破坏必崩溃
    'jmp': 9,        # 无条件跳转
    'jcc': 8,        # 条件跳转 (je, jne, jl, jg, etc.)

    # 中高敏感 (6-8分)
    'push': 8,       # 压栈，破坏栈结构
    'pop': 8,        # 弹栈，破坏栈结构
    'mov_mem': 7,    # 内存写操作
    'lea': 6,        # 地址计算

    # 中等敏感 (4-6分)
    'div': 6,        # 除法，可能除零
    'idiv': 6,       # 有符号除法
    'mov_reg': 4,    # 寄存器间移动

    # 低敏感 (0-4分)
    'add': 3,        # 算术运算
    'sub': 3,
    'mul': 3,
    'xor': 2,        # 逻辑运算
    'and': 2,
    'or': 2,
    'nop': 0,        # 空操作
}
```

#### 2.2.2 寄存器敏感性 (RegScore)

```python
# 寄存器敏感性评分 (0-10分)
REGISTER_CRASH_SCORE = {

    'rsp': 10,  # 栈指针
    'rbp': 9,   # 帧指针

    # 高敏感 - 常用于地址计算
    'rax': 6,   # 返回值/累加器
    'rbx': 5,   # 基址寄存器
    'rcx': 5,   # 计数器
    'rdx': 5,   # 数据寄存器

    # 中等敏感 - 通用寄存器
    'rsi': 4,   # 源索引
    'rdi': 4,   # 目标索引
    'r8-r15': 3,  # 扩展寄存器

    # 低敏感 - 浮点/向量寄存器

    'ymm0-ymm15': 2,  # AVX 寄存器
}
```

#### 2.2.3 内存操作模式 (MemScore)

```python
# 内存操作模式评分 (0-10分)
MEMORY_MODE_CRASH_SCORE = {
    # 写操作 - 高风险
    'store_indirect': 10,   # 间接寻址写 mov [rax], rbx
    'store_indexed': 9,     # 索引寻址写 mov [rax+rbx*4], rcx
    'store_direct': 7,      # 直接寻址写 mov [addr], rax

    # 读操作 - 中等风险
    'load_indirect': 6,     # 间接寻址读
    'load_indexed': 5,      # 索引寻址读
    'load_direct': 3,       # 直接寻址读

    # 栈操作 - 特殊处理
    'stack_write': 8,       # 栈写入
    'stack_read': 4,        # 栈读取

    # 非内存操作
    'no_memory': 1,
}
```

#### 2.2.4 历史崩溃率 (HistoryScore)

基于已有实验数据的统计：

```python
def calculate_history_score(inst_type, pc, historical_data):
    """
    基于历史实验数据计算崩溃概率评分

    Args:
        inst_type: 指令类型
        pc: 指令地址
        historical_data: 历史实验结果 DataFrame

    Returns:
        score: 0-10 的评分
    """
    # 按指令类型统计
    type_data = historical_data[historical_data['op'] == inst_type]
    if len(type_data) > 0:
        crash_count = len(type_data[type_data['result'].isin([
            'Recrash', 'C-Masked', 'C-SDC'
        ])])
        crash_rate = crash_count / len(type_data)
        return crash_rate * 10
    return 5  # 默认中等分数
```

```cpp
// crashProbabilityAnalyzer.cpp

struct InstInfo {
    ADDRINT pc;           // 指令地址
    string mnemonic;      // 指令助记符
    string dest_reg;      // 目标寄存器
    string src_regs;      // 源寄存器列表
    bool is_memory_write; // 是否内存写
    bool is_memory_read;  // 是否内存读
    bool is_stack_op;     // 是否栈操作
    bool is_control_flow; // 是否控制流指令
    UINT64 exec_count;    // 动态执行次数
    float crash_score;    // 崩溃概率评分
};

// 分析函数
VOID AnalyzeInstruction(INS ins) {
    InstInfo info;
    info.pc = INS_Address(ins);
    info.mnemonic = INS_Mnemonic(ins);

    // 检测内存写操作
    info.is_memory_write = INS_IsMemoryWrite(ins);

    // 检测控制流指令
    info.is_control_flow = INS_IsControlFlow(ins);

    // 检测栈操作
    info.is_stack_op = INS_IsStackWrite(ins) || INS_IsStackRead(ins);

    // 获取目标寄存器
    for (UINT32 i = 0; i < INS_MaxNumWRegs(ins); i++) {
        REG reg = INS_RegW(ins, i);
        info.dest_reg += REG_StringShort(reg) + ",";
    }

    // 计算崩溃概率评分
    info.crash_score = CalculateCrashScore(info);

    // 保存到 catalog
    SaveToCatalog(info);
}
```

### 3.2 修改 `InstPoolMaker.py`

```python
# InstPoolMaker.py 新增函数

def generate_crash_weighted_pool(
    catalog_csv_file,
    pool_csv_file,
    num_samples=2000,
    crash_weight_mode='high'  # 'high', 'medium', 'low', 'balanced'
):
    """
    生成基于崩溃概率加权的指令池

    Args:
        catalog_csv_file: 包含崩溃评分的 catalog
        pool_csv_file: 输出的指令池
        num_samples: 采样数量
        crash_weight_mode: 权重模式
            - 'high': 优先选择高崩溃概率指令
            - 'medium': 中等崩溃概率
            - 'low': 低崩溃概率
            - 'balanced': 均匀分布
    """
    import pandas as pd
    import numpy as np

    # 读取带有崩溃评分的 catalog
    df = pd.read_csv(catalog_csv_file)

    # 确保有 crash_score 列
    if 'crash_score' not in df.columns:
        df['crash_score'] = calculate_crash_scores(df)

    # 根据模式计算采样权重
    if crash_weight_mode == 'high':
        # 指数加权，高分获得更高权重
        weights = np.exp(df['crash_score'] / 2)
    elif crash_weight_mode == 'medium':
        # 正态分布，中间分数权重最高
        weights = np.exp(-((df['crash_score'] - 5) ** 2) / 8)
    elif crash_weight_mode == 'low':
        # 反向指数加权
        weights = np.exp((10 - df['crash_score']) / 2)
    else:  # balanced
        weights = np.ones(len(df))

    # 归一化权重
    weights = weights / weights.sum()

    # 加权随机采样
    selected_indices = np.random.choice(
        len(df), size=num_samples, replace=True, p=weights
    )

    # 生成指令池
    selected_rows = df.iloc[selected_indices]

    # 保存到 pool 文件
    pool_data = []
    for _, row in selected_rows.iterrows():
        pool_data.append([
            row['regmm'] if pd.notna(row['regmm']) else '',
            row['reg'] if pd.notna(row['reg']) else '',
            str(int(row['pc'], 16)) if isinstance(row['pc'], str) else str(row['pc']),
            str(np.random.randint(0, min(int(row['exec_count']), 1024)))
        ])

    with open(pool_csv_file, 'w', newline='') as f:
        writer = csv.writer(f)
        writer.writerows(pool_data)

    print(f"已生成 {num_samples} 条高崩溃概率指令，保存到 {pool_csv_file}")
    return pool_csv_file


def calculate_crash_scores(df):
    """
    为 DataFrame 中的每条指令计算崩溃概率评分
    """
    scores = []

    for _, row in df.iterrows():
        score = 0

        # 1. 指令类型评分
        mnemonic = row.get('mnemonic', '').lower()
        score += get_inst_type_score(mnemonic) * 0.3

        # 2. 寄存器评分
        reg = row.get('reg', '') or row.get('regmm', '')
        score += get_register_score(reg) * 0.25

        # 3. 内存操作评分
        is_mem_write = row.get('is_memory_write', False)
        is_stack = row.get('is_stack_op', False)
        score += get_memory_score(is_mem_write, is_stack) * 0.25

        # 4. 控制流评分
        is_cf = row.get('is_control_flow', False)
        if is_cf:
            score += 2.0

        scores.append(min(score, 10))  # 限制最大分数为 10

    return scores


def get_inst_type_score(mnemonic):
    """指令类型评分"""
    HIGH_CRASH = ['call', 'ret', 'jmp', 'je', 'jne', 'jl', 'jg', 'jle', 'jge']
    MED_CRASH = ['push', 'pop', 'mov', 'lea', 'div', 'idiv']

    if any(m in mnemonic for m in HIGH_CRASH):
        return 9
    elif any(m in mnemonic for m in MED_CRASH):
        return 6
    else:
        return 3


def get_register_score(reg):
    """寄存器敏感性评分"""
    if not reg:
        return 3
    reg = reg.lower()
    if reg in ['rip', 'eip']:
        return 10
    elif reg in ['rsp', 'esp', 'rbp', 'ebp']:
        return 9
    elif reg in ['rax', 'eax', 'rbx', 'ebx', 'rcx', 'ecx', 'rdx', 'edx']:
        return 5
    else:
        return 3


def get_memory_score(is_mem_write, is_stack):
    """内存操作评分"""
    if is_mem_write and is_stack:
        return 9
    elif is_mem_write:
        return 7
    elif is_stack:
        return 5
    else:
        return 2
```

```python
from sklearn.ensemble import RandomForestClassifier

def train_crash_predictor(historical_data):

    训练崩溃预测模型

        'inst_type_encoded',
        'dest_reg_encoded',

        'is_stack_op',

    X = historical_data[features]

    model = RandomForestClassifier(n_estimators=100)

    return model

def predict_crash_probability(model, inst_info):
    """
    预测指令的崩溃概率
    """
    return model.predict_proba([inst_info])[0][1]
```

```python
class AdaptiveCrashSelector:
    """
    在线自适应指令选择器
    根据实时实验结果调整选择策略
    """

    def __init__(self, initial_scores):
        self.scores = initial_scores
        self.history = []

    def update(self, inst_info, result):
        """
        根据实验结果更新评分
        """
        key = (inst_info['mnemonic'], inst_info['dest_reg'])

        if result in ['Recrash', 'C-Masked', 'C-SDC']:
            # 崩溃了，增加该类型的评分
            self.scores[key] = min(10, self.scores.get(key, 5) + 0.5)
        else:
            # 没崩溃，降低评分
            self.scores[key] = max(0, self.scores.get(key, 5) - 0.2)

    def select_next(self, catalog):
        """
        选择下一个注入目标
        """
        # 使用 Thompson Sampling 或 UCB 算法
        return self._thompson_sampling(catalog)
```

## 7. 总结

本设计方案提供了一套完整的高崩溃概率指令选择策略：

1. **多维度评分模型**：综合指令类型、寄存器、内存操作、历史数据
2. **加权随机采样**：基于评分的概率采样
3. **可配置权重模式**：支持高/中/低崩溃概率选择
4. **历史数据驱动**：利用已有实验结果优化选择
5. **在线自适应**：实时调整选择策略

实现这套策略后，预期可以将崩溃率提高 50-100%，显著提升恢复策略的测试效率。
